# Structural Long-Timeseries Learning

## Abstract
Traditional Transformers are "greedy": they optimize for the next token (+1$), often overfitting to local noise and missing long-term structural phase transitions. 

This repository implements a **Non-Greedy Structural Learning** approach. By separating "momentum" (short-term) from "structure" (long-term) using orthogonal representations, we aim to predict endogenous regime shifts (financial crises) in non-stationary time series.

## Setup & Usage (uv)
This project uses `uv` for high-performance dependency management.

```bash
# Install dependencies
uv sync

# Run training pipeline
uv run python src/train.py
```

import torch
import torch.nn as nn
import torch.nn.functional as F

class DialecticalLoss(nn.Module):
    """
    Función de pérdida compuesta para DINN (Dialectical Neural Networks).
    Implementa:
    1. MSE (Precisión)
    2. Penalización Ortogonal (Lucha de contrarios)
    3. Penalización de Acumulación (Cantidad a Cualidad)
    """
    def __init__(self, lambda_orth=0.1, lambda_accumulation=0.01):
        super().__init__()
        self.mse = nn.MSELoss()
        self.lambda_orth = lambda_orth
        self.lambda_acc = lambda_accumulation

    def orthogonal_regularization(self, h_thesis, h_antithesis):
        """
        Fuerza a la Tesis y Antítesis a aprender características disjuntas.
        Minimiza el coseno cuadrado de similitud.
        """
        # Normalizar vectores
        h_t_norm = F.normalize(h_thesis, p=2, dim=1)
        h_a_norm = F.normalize(h_antithesis, p=2, dim=1)
        
        # Calcular coseno similitud
        cosine = torch.sum(h_t_norm * h_a_norm, dim=1)
        
        # Queremos que sea 0 (ortogonal), así que minimizamos el cuadrado
        return torch.mean(cosine ** 2)

    def forward(self, pred, target, h_thesis, h_antithesis):
        # 1. Error de predicción estándar
        loss_pred = self.mse(pred, target)
        
        # 2. Lucha de Contrarios (Regularización)
        loss_orth = self.orthogonal_regularization(h_thesis, h_antithesis)
        
        # Total
        return loss_pred + (self.lambda_orth * loss_orth)

import torch
import torch.nn as nn

class AufhebenLayer(nn.Module):
    """
    Capa de Síntesis Dialéctica.
    En lugar de concatenar o sumar, utiliza una interacción bilineal
    para modelar cómo la antítesis transforma la tesis.
    """
    def __init__(self, input_dim, output_dim):
        super().__init__()
        # Proyecciones para reducir dimensionalidad antes de la síntesis
        self.proj_thesis = nn.Linear(input_dim, output_dim)
        self.proj_antithesis = nn.Linear(input_dim, output_dim)
        
        # Compuerta de Regulación (Gating Mechanism)
        self.gate = nn.Sequential(
            nn.Linear(output_dim * 2, 1),
            nn.Sigmoid()
        )
        
        self.norm = nn.LayerNorm(output_dim)

    def forward(self, z_thesis, z_antithesis):
        # 1. Proyecciones
        t = self.proj_thesis(z_thesis)
        a = self.proj_antithesis(z_antithesis)
        
        # 2. Interacción No Lineal (Producto de Hadamard - Síntesis)
        # Representa la penetración mutua de los opuestos
        synthesis = t * a 
        
        # 3. Gating Dinámico: ¿Estamos en periodo normal o de crisis?
        combined = torch.cat([t, a], dim=-1)
        w = self.gate(combined)
        
        # 4. Salida Residual (Negación de la Negación)
        # Preserva la identidad (t) pero modificada por la síntesis
        out = (w * t) + ((1 - w) * synthesis)
        
        return self.norm(out)

import pandas as pd
import numpy as np
import yaml
import argparse
from pathlib import Path

# --- UPDATE: Ruta correcta para NeurIPS structure ---
CONFIG_PATH = Path("configs/benchmarks/finance.yaml")

class DataProcessor:
    """
    Handles the transformation from Raw Nominal Data to Real Structural Features.
    Calculates DRAWDOWN for crisis detection.
    """
    
    def __init__(self, config_path: Path):
        self.config_path = config_path
        if not self.config_path.exists():
             raise FileNotFoundError(f"Config not found at: {self.config_path}")
             
        with open(self.config_path, 'r') as f:
            self.config = yaml.safe_load(f)

    def load_raw_data(self, dataset_name: str) -> pd.DataFrame:
        if dataset_name not in self.config['datasets']:
            raise ValueError(f"Dataset '{dataset_name}' not found in config.")
            
        raw_path = Path(self.config['datasets'][dataset_name]['save_path'])
        if not raw_path.exists():
            raise FileNotFoundError(f"Raw file not found: {raw_path}. Run loader first.")
            
        print(f"   [IO] Loading raw data: {raw_path}")
        return pd.read_csv(raw_path, index_col=0, parse_dates=True)

    def deflate_nominal_series(self, df: pd.DataFrame, cpi_col: str = 'CPI_ADJUSTMENT') -> pd.DataFrame:
        if cpi_col not in df.columns:
            print("   [WARN] CPI column not found. Skipping inflation adjustment.")
            return df
            
        cpi_base = df[cpi_col].iloc[-1]
        print(f"   [MATH] Deflating series to current USD (Base CPI: {cpi_base:.2f})")
        
        exclude_cols = [cpi_col, 'RISK_FREE_RATE', 'CAPACITY_UTIL', 'LIQUIDITY_PREMIUM', 'IMPLIED_VOLATILITY', 'CPI', 'RECESSION']
        
        df_real = df.copy()
        for col in df.columns:
            if col not in exclude_cols and not col.endswith('_REAL'):
                new_col_name = f"{col}_REAL"
                df_real[new_col_name] = df[col] * (cpi_base / df[cpi_col])
        
        return df_real

    def calculate_drawdown(self, df: pd.DataFrame, target_col: str) -> pd.DataFrame:
        """Calculates the % decline from the historical peak."""
        if target_col not in df.columns:
            return df
        
        # 1. Rolling Max (Historical Peak)
        rolling_peak = df[target_col].cummax()
        
        # 2. Drawdown formula
        df['DRAWDOWN'] = (df[target_col] - rolling_peak) / rolling_peak
        
        print(f"   [MATH] Calculated Structural Drawdown for {target_col}")
        return df

    def run(self):
        print("Starting Data Processing Pipeline...")
        
        for name, params in self.config['datasets'].items():
            print(f"\nProcessing Dataset: {name}")
            
            # 1. Load
            df = self.load_raw_data(name)
            
            # 2. Deflate
            df = self.deflate_nominal_series(df, cpi_col='CPI_ADJUSTMENT')
            
            # 3. Calculate Crisis Metrics (Drawdown)
            target = 'MARKET_VALUATION_REAL' if 'MARKET_VALUATION_REAL' in df.columns else None
            if target:
                df = self.calculate_drawdown(df, target_col=target)
            
            # 4. Save
            processed_path = Path(params['save_path'].replace('raw', 'processed'))
            processed_path.parent.mkdir(parents=True, exist_ok=True)
            df.to_csv(processed_path)
            print(f"   [SUCCESS] Saved processed data to: {processed_path}")

if __name__ == "__main__":
    # Si se corre directo, usa la constante global
    processor = DataProcessor(CONFIG_PATH)
    processor.run()

def plot_corporate_profits(df, output_dir):

        return

def plot_global_growth(df, output_dir):

def main(dataset_name):
    processed_path = Path(f"data/processed/{dataset_name}.csv")
    output_dir = Path(f"out/{dataset_name}")
    output_dir.mkdir(parents=True, exist_ok=True)
    
    if not processed_path.exists():
        print(f"❌ Data file not found: {processed_path}. Run processor first.")
        return

    print(f"📊 Visualizing: {dataset_name}...")
    df = pd.read_csv(processed_path, index_col=0, parse_dates=True)
    
    # DISPATCHER LIMPIO
    if dataset_name == 'corporate_profits':
        plot_corporate_profits(df, output_dir)
    elif dataset_name == 'global_growth':
        plot_global_growth(df, output_dir)
    elif dataset_name == 'capital_cycle':
        plot_capital_cycle(df, output_dir)
    else:
        print(f"No specific plot defined for {dataset_name}.")

if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--dataset", type=str, required=True, help="Dataset name from config")
    args = parser.parse_args()
    main(args.dataset)

import yaml
import pandas_datareader.data as web
import pandas as pd
import os
from datetime import datetime

CONFIG_PATH = 'config/data_config.yaml'

def load_config():
    if not os.path.exists(CONFIG_PATH):
        raise FileNotFoundError(f"❌ No config: {CONFIG_PATH}")
    with open(CONFIG_PATH, 'r') as file:
        return yaml.safe_load(file)

def fetch_dataset(name, params):
    print(f"\n⬇️  [EXTRACT] Descargando dataset: '{name}'")
    indicators = params['indicators']
    
    try:
        # 1. Descarga RAW (Nominal)
        df = web.DataReader(list(indicators.values()), 'fred', params['start_date'], datetime.now())
        
        # 2. Renombrar
        df.rename(columns={v: k for k, v in indicators.items()}, inplace=True)
        
        # 3. Resampling básico
        df = df.resample(params['frequency']).mean().ffill()
        
        # 4. Guardar en RAW
        raw_path = params['save_path'] # e.g., data/raw/macro_structural.csv
        os.makedirs(os.path.dirname(raw_path), exist_ok=True)
        df.to_csv(raw_path)
        print(f"✅ Guardado RAW en: {raw_path}")
        
    except Exception as e:
        print(f"❌ Error descargando '{name}': {e}")

if __name__ == "__main__":
    config = load_config()
    for name, params in config['datasets'].items():
        fetch_dataset(name, params)

import argparse
import yaml
import sys
from pathlib import Path

# --- SETUP DE PATHS ---
BASE_DIR = Path(__file__).resolve().parents[3]
sys.path.append(str(BASE_DIR))

# --- IMPORTS DIRECTOS (La clave del cambio) ---
# En lugar de os.system, traemos el código aquí
from src.benchmarks.finance import loader
from src.benchmarks.finance.processor import DataProcessor
from src.benchmarks.finance.visualization import main as visualize_dataset

def load_config(config_path):
    with open(config_path, 'r') as f:
        return yaml.safe_load(f)

def run_pipeline():
    parser = argparse.ArgumentParser(description="NeurIPS Finance Benchmark Pipeline")
    parser.add_argument("--step", choices=["all", "load", "process", "viz"], default="all")
    args = parser.parse_args()

    # Configuración Centralizada
    config_path = BASE_DIR / "configs/benchmarks/finance.yaml"
    if not config_path.exists():
        print(f"❌ Config no encontrado en: {config_path}")
        return

    config = load_config(config_path)
    print(f"📜 Configuración cargada: {config_path.name}")

    # --- PASO 1: LOADER (Puro y Declarativo) ---
    if args.step in ["all", "load"]:
        print("\n=== [STEP 1] DATA INGESTION ===")
        for name, params in config['datasets'].items():
            loader.fetch_dataset(name, params)

    # --- PASO 2: PROCESSOR (Inyección de Dependencia) ---
    if args.step in ["all", "process"]:
        print("\n=== [STEP 2] DATA PROCESSING ===")
        # Instanciamos la clase directamente pasando el path del config
        # Esto es mucho más robusto que llamar a un script externo
        processor = DataProcessor(config_path)
        processor.run()

    # --- PASO 3: VISUALIZATION (Llamada a Función) ---
    if args.step in ["all", "viz"]:
        print("\n=== [STEP 3] VISUALIZATION ===")
        # Iteramos sobre lo que está declarado en el YAML
        for ds_name in config['datasets'].keys():
            print(f">> Plotting {ds_name}...")
            # Llamamos a la función main del visualizador directamente
            try:
                visualize_dataset(ds_name)
            except Exception as e:
                print(f"⚠️  Could not plot {ds_name}: {e}")

if __name__ == "__main__":
    run_pipeline()
